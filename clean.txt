LiquidCrystal_I2C lcd(0x27, 16, 2);

/* ---------- PMS5003 ---------- */
HardwareSerial PMS(2);

#define RELAY_PIN 23   // Active LOW relay

/* ---------- WEB SERVER ---------- */
WebServer server(80);

/* ---------- VARIABLES ---------- */
uint16_t pm1_0 = 0, pm2_5 = 0, pm10 = 0;
uint8_t buffer[32];
bool motorState = false;
bool manualMode = false; // false = Auto Mode, true = Manual Mode

/* ---------- READ PMS DATA ---------- */
bool readPMS() {
  if (PMS.available() < 32) return false;
  if (PMS.read() != 0x42) return false;
  if (PMS.read() != 0x4D) return false;

  buffer[0] = 0x42;
  buffer[1] = 0x4D;

  if (PMS.readBytes(&buffer[2], 30) != 30) return false;

  uint16_t sum = 0;
  for (int i = 0; i < 30; i++) sum += buffer[i];

  uint16_t checksum = (buffer[30] << 8) | buffer[31];
  if (sum != checksum) return false;

  pm1_0 = (buffer[4] << 8) | buffer[5];
  pm2_5 = (buffer[6] << 8) | buffer[7];
  pm10  = (buffer[8] << 8) | buffer[9];

  return true;
}

/* ---------- ROUTES ---------- */
void handleRoot() {
  server.send_P(200, "text/html", PAGE);
}

void handleData() {
  String json = "{";
  json += "\"pm1_0\":" + String(pm1_0) + ",";
  json += "\"pm2_5\":" + String(pm2_5) + ",";
  json += "\"pm10\":"  + String(pm10) + ",";
  json += "\"motor\":" + String(motorState ? 1 : 0) + ",";
  json += "\"manual\":" + String(manualMode ? 1 : 0);
  json += "}";
  server.send(200, "application/json", json);
}

void handleSetMode() {
  if (server.hasArg("manual")) {
    manualMode = (server.arg("manual").toInt() == 1);
  }
  handleData();
}

void handleToggleMotor() {
  if (manualMode && server.hasArg("state")) {
    int state = server.arg("state").toInt();
    motorState = (state == 1);
    digitalWrite(RELAY_PIN, motorState ? LOW : HIGH); // Active LOW mapping
  }
  handleData();
}

/* ---------- SETUP ---------- */
void setup() {
  Serial.begin(115200);
  delay(1500); 

  WiFi.softAP(AP_SSID, AP_PASS);

  pinMode(RELAY_PIN, OUTPUT);
  digitalWrite(RELAY_PIN, HIGH); // Start with Motor OFF

  lcd.init();
  lcd.backlight();
  lcd.print("System Starting");

  PMS.begin(9600, SERIAL_8N1, RX_PIN, TX_PIN);

  server.on("/", handleRoot);
  server.on("/data", handleData);
  server.on("/setMode", handleSetMode);
  server.on("/toggleMotor", handleToggleMotor);
  server.begin();

  lcd.clear();
  lcd.print("WiFi Ready:");
  lcd.setCursor(0,1);
  lcd.print(AP_SSID);
}

/* ---------- MAIN LOOP ---------- */
void loop() {
  server.handleClient();

  if (readPMS()) {
    /* ---------- AUTOMATED CONTROL ---------- */
    if (!manualMode) {
      if (pm2_5 > 200) {
        digitalWrite(RELAY_PIN, LOW); // Turn Motor ON
        motorState = true;
      } else {
        digitalWrite(RELAY_PIN, HIGH); // Turn Motor OFF
        motorState = false;
      }
    }

    /* ---------- LCD DISPLAY WITH UNITS ---------- */
    lcd.clear();
    lcd.setCursor(0,0);
    lcd.print("PM2.5:");
    lcd.print(pm2_5);
    lcd.print("ug/m3");

    lcd.setCursor(0,1);
    lcd.print("PM10 :");
    lcd.print(pm10);
    lcd.print("ug/m3");
  }
  delay(1000);
}
